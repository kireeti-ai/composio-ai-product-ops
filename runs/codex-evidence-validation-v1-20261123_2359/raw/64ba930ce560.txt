      updateState({ status: "loading" });

      try {
        var response = await fetch("https://api.clay.com/v3/me", {
          credentials: "include",

fs-toc-offsettop="8rem" fs-toc-element="contents" class="rich-text w-richtext"><h1><strong>Account settings</strong></h1><p>Your account settings allow you to keep your personal data up to date and ensure your account remains secure. Below are the steps for updating key details like your profile picture, name, and password.</p><h2><strong>Update your profile picture</strong></h2><p>To update your profile picture:</p><ul><li>Go to <code>Settings</code> and select <code>Account</code> from the left-hand menu.</li><li>In the <code>Your details</code> tab under <code>Your profile</code>, click <code>Upload new picture</code> to upload an image or choose an icon.<ul><li>Please ensure the image is in png, jpg, jpeg, or gif format with a max size of 5MB.</li></ul></li><li>Use the <code>Delete</code> button if you wish to remove your current profile picture.</li><li>Click <code>Save</code> to confirm your changes.</li></ul><h2><strong>Update your name</strong></h2><p>To update your account name:</p><ul><li>Go to <code>Settings</code> and select <code>Account</code>.</li><li>Under the <code>Your details</code> tab, edit your name in the <code>Name</code> field.</li><li>Click <code>Save</code> to ensure your changes are updated.</li></ul><h2><strong>Change your password</strong></h2><p>To change your account password:</p><ul><li>Visit your <code>Settings</code> and head to the <code>Account</code> section.</li><li>Open the <code>Security</code> tab under <code>Your profile</code>.</li><li>Click <code>Change password</code>. A magic link will be sent to your registered email address.<ul><li><em>Note: This option is available only if you signed in using a password and not via an authentication service such as Google or SSO.</em></li></ul></li><li>Follow the instructions in the email to securely update your password.</li></ul><h2><strong>Clay API key access</strong></h2><p>Your Clay API key enables Clay-specific integrations and external connections. To manage your API key:</p><ul><li>Go to <code>Settings</code> &gt;<code>Your profile</code> &gt; <code>API key</code>.</li><li>Use the following options:<ul><li><code>Copy</code> the key for application use.</li><li><code>Regenerate</code> the key if compromised.</li></ul></li></ul><h2><strong>Delete your account</strong></h2><p>You can permanently delete your Clay account through your account settings. Before proceeding, please note:</p><p><strong>Requirements:</strong></p><ul><li>You must be a <strong>member</strong> (not admin) of all your workspaces, OR</li><li>You must be the <strong>only member</strong> in your workspace</li></ul><p><strong>To delete your account:</strong></p><ul><li>Go to <code>Settings</code> and select <code>Account</code> from the left-hand menu.</li><li>Scroll to the <code>Account deletion</code> section at the bottom of the page.</li><li>Click the <code>Delete account</code> button.</li><li>Complete the email verification step to confirm your request.</li></ul><p><strong>What happens when you delete your account:</strong></p><ul><li>Your deletion request is processed and logged for audit purposes.</li><li>Your API keys are deleted.</li><li>Your account email is updated to a deleted variant.</li><li>For any workspaces affected by your account deletion, workspace admins will receive email notifications.</li><li>Your private app account and Stripe customer information are deleted to prevent unexpected charges.</li></ul><div class="w-embed"><div style="background-color: #FDF0F9; padding: 16px; border-radius: 8px; border: 1px solid #F7CCE2; font-family: Arial, sans-serif;"><strong>Important:</strong> Account deletion is permanent and cannot be undone. While your data is marked for deletion and critical billing/authentication records are removed immediately, full data removal from our database may take additional time.</div></div><h1><strong>Workspace settings</strong></h1><p>Workspace settings give you control over key aspects of your workspace, such as its name, profile picture, and billing email. These settings ensure your workspace is easily identifiable and that billing communications reach the correct contact. Below are the steps for updating key workspace details.</p><h2><strong>Workspace picture</strong></h2><p>To update your workspace picture:</p><ul><li>Navigate to <code>Settings</code> &gt; <code>Workspace settings</code>.</li><li>Click <code>Upload new picture</code> to upload an image or choose an icon.<ul><li>Please ensure the image is in png, jpg, jpeg, or gif format with a max size of 5MB.</li></ul></li><li>Use the <code>Delete</code> button if you wish to remove your current profile picture.</li><li>Click <code>Save</code> to apply your changes.</li></ul><h2><strong>Workspace name</strong></h2><p>To update your workspace name:</p><ul><li>Go to <code>Settings</code> &gt; <code>Workspace settings</code>.</li><li>Update the <code>Workspace name</code> field with your desired name.<ul><li>This name will be displayed across your workspace and should be accessible for team identification.</li></ul></li><li>Click <code>Save</code> to confirm your changes.</li></ul><h2><strong>Billing email</strong></h2><p>To update your billing email:</p><ul><li>In <code>Workspace settings</code>, edit the <code>Billing email</code> field to update the email address used for all billing-related communication.</li><li>Click <code>Save</code> to ensure the new email is recorded.</li></ul><h2><strong>Delete your workspace</strong></h2><p>You can permanently delete your Clay workspace through workspace settings. This action is permanent and cannot be undone.</p><p><strong>Requirements:</strong></p><ul><li>The workspace must have <strong>no pending invites</strong>.</li><li>You must be the <strong>only member</strong> in the workspace.</li></ul><p><strong>To delete your workspace:</strong></p><ul><li>Navigate to <code>Settings</code> &gt; <code>Workspace settings</code>.</li><li>Scroll to the <code>Delete workspace</code> section at the bottom of the page.</li><li>Click the <code>Delete workspace</code> button.</li><li>Complete the email verification step to confirm your request.</li></ul><p><strong>What happens when you delete your workspace:</strong></p><ul><li>Your deletion request is processed and logged for audit purposes.</li><li>The workspace owner is updated to prevent access.</li><li>Your private app account and Stripe customer information for the workspace are deleted to prevent unexpected charges.</li><li>Workspace admins receive email notifications about the deletion.</li></ul><div class="w-embed"><div style="background-color: #FDF0F9; padding: 16px; border-radius: 8px; border: 1px solid #F7CCE2; font-family: Arial, sans-serif;"><strong>Important:</strong> Workspace deletion is permanent and cannot be undone. All workspace data, tables, and configurations will be permanently removed.</div></div><h1><strong>Managing team members</strong></h1><h2><strong>Roles and permissions</strong></h2><p>Clay offers three user roles with different permission levels to help manage your workspace effectively.</p><h3><strong>Admin</strong></h3><p><strong>Admins</strong> have full control over the workspace and can manage both resources and team membership.</p><p><strong>Capabilities:</strong></p><ul><li>Manage all workspace settings and resources.</li><li>Invite and remove team members.</li><li>Assign and update user roles.</li><li>Access billing and workspace configuration.</li></ul><h3><strong>Editor</strong></h3><p><strong>Editors</strong> are standard users focused on building and working within workspace resources.</p><p><strong>Capabilities:</strong></p><ul><li>Create and edit tables, workflows, and integrations.</li><li>Update records in tables.</li><li>Delete tables they own.</li><li>Add or modify columns and data sources.</li></ul><p><strong>Editors cannot:</strong></p><ul><li>Add, remove, or change team members' roles.</li><li>Modify billing settings or purchase credits.</li></ul><h3><strong>Viewer</strong></h3><p><strong>Viewers</strong> have read-only access to workspace content.</p><div class="w-embed"><div style="background-color: #FDFBF0; padding: 16px; border-radius: 8px; border: 1px solid #F4E8C1; font-family: Arial, sans-serif;"> <strong>Note:</strong> The Viewer role is available on the Enterprise plan only. </div></div><h2><strong>Add a team member to your workspace</strong></h2><p>To invite a new member to your workspace:</p><ul><li>Go to <code>Settings</code> &gt; <code>Team</code>.</li><li>Click the <code>+ Invite</code> button in the top-right corner.</li><li>Enter the email address of the person you want to invite.</li><li>Select the appropriate role (Editor or Admin) from the dropdown.</li><li>Click <code>Send invite</code>.</li></ul><p>The invited person will receive an email to join the workspace with the specified role.</p><h2><strong>Change a team member's role</strong></h2><p>To update a member's role:</p><ul><li>Navigate to <code>Settings</code> &gt; <code>Team</code>.</li><li>Locate the team member whose role you want to update.</li><li>Use the dropdown menu next to their name to select the desired role.</li><li>Changes are applied immediately.</li></ul><h2><strong>Remove a team member</strong></h2><p>To remove a member from your workspace:</p><ul><li>Go to <code>Settings</code> &gt; <code>Team</code>.</li><li>Find the member you wish to remove.</li><li>Click the <code>…</code> menu next to their name and select <code>Remove member</code>.</li><li>In the <code>Remove member?</code> dialog, click <code>Remove</code> to confirm.</li></ul><h1><strong>Connections</strong></h1><p>The <code>Connections</code> settings allow you to manage integrations with external services. Each integration is represented as an <code>account</code> with its own authentication and configuration.</p><p>You can connect multiple accounts to the same integration for added flexibility, designate default accounts for specific integrations, and modify, reconnect, or remove accounts as needed.</p><h2><strong>What is an integration account?</strong></h2><p>An <strong>integration account</strong> is a configured connection between your workspace and an external service, such as an API or data provider. Each account enables your workspace to authenticate and interact with the service to perform specific tasks or access features.</p><ul><li><code>Account / Key Name</code>: A user-defined name to identify the account, such as "Marketing HubSpot Account" or "Development Anthropic Key." This helps distinguish it from other accounts in the same service.</li><li><code>Account Credentials</code>: The authentication details (e.g., API keys or OAuth tokens) required to connect securely to the external service. Credentials can be tested or updated if they become invalid.</li><li><code>Default Status</code>: An optional setting that makes the account the default choice for its service, streamlining its use in workflows.</li></ul><h2><strong>Types of accounts</strong></h2><p><strong>User-managed accounts:</strong></p><ul><li>Configured by users with their own credentials, such as personal API keys or OAuth tokens.</li><li>Ideal for integrations that require team- or project-specific access.</li></ul><p><strong>Clay-provided accounts:</strong></p><ul><li>Provided by Clay and configured for public use.</li><li>These accounts use workspace credits instead.</li><li>Labeled <code>Clay-provided</code> in the <code>Account type</code> column of the <code>Connections</code> page, and often pre-set as default.</li></ul><h2><strong>Add a new integration account</strong></h2><p>To add a new account for an integration:</p><ul><li>Navigate to <code>Settings</code> &gt; <code>Connections</code>.</li><li>Click <code>+ Add connection</code> in the top-right corner.</li><li>Use the search bar to locate the service you want to integrate with.</li><li>Follow the on-screen instructions to authenticate and configure the connection.</li><li>Once completed, the account will appear under the corresponding service in the <code>Connections</code> list.</li></ul><h2><strong>Managing existing accounts</strong></h2><h3><strong>View your integration accounts</strong></h3><ul><li>Navigate to <code>Settings</code> &gt; <code>Connections</code>.</li><li>Select a service (e.g., HubSpot or Anthropic) to see all associated accounts.</li><li>Use the search bar to quickly locate a specific account.</li></ul><h3><strong>Edit your integration accounts</strong></h3><ul><li>Select the service and click the <code>…</code> menu next to an account.</li><li>Choose <code>Edit</code> to:<ul><li>Update the <code>Key Name</code>.</li><li>Modify or re-enter credentials.</li></ul></li><li>Toggle <code>Set as Default</code> to make this account the default.</li><li>Click <code>Save</code> to apply changes.</li></ul><h3><strong>Testing credentials</strong></h3><p>To ensure your account credentials are valid and the integration is functioning correctly:</p><ul><li>Navigate to <code>Settings</code> &gt; <code>Connections</code>.</li><li>Select the service (e.g., Anthropic) to view its associated accounts.</li><li>Locate the account you want to test.</li><li>Click the <code>…</code> menu next to the account and select <code>Test Connection</code>.</li></ul><h3><strong>Setting default accounts</strong></h3><p>A <strong>default account</strong> is automatically selected for workflows or integrations where no specific account is specified. To set an account as default:</p><ul><li>Navigate to <code>Settings</code> &gt; <code>Connections</code> and select the desired service (e.g., Anthropic).</li><li>Locate the account you wish to set as the default.</li><li>Click the <code>…</code> menu next to the account, and select <code>Set as Default</code>.</li><li>The account will now display a <code>default</code> label to indicate its status.</li></ul><h3><strong>Deleting accounts</strong></h3><ul><li>Navigate to the service in the <code>Connections</code> section.</li><li>Click the <code>…</code> menu next to the account and select <code>Delete</code>.</li><li>Confirm the deletion.</li></ul><h3><strong>Clay-provided accounts</strong></h3><p>Clay-provided accounts simplify access to external services by providing pre-configured integrations that use workspace credits instead of requiring individual credentials.</p><p>For some integrations, only Clay-provided accounts are supported. For others, you will need to provide your own API key.</p><h1><strong>Plans and billing</strong></h1><h2><strong>Understanding actions and data credits</strong></h2><p>Clay's pricing is built on two separate usage meters:</p><ul><li><strong>Actions</strong> measure the work Clay orchestrates on your behalf — enriching data, running AI research, syncing to a CRM, exporting to CSV, sending to a sequencer, and more. Each enrichment or execution task consumes 1 action. Actions are a background meter included in your plan; 90% of customers will never hit their actions limit.</li><li><strong>Data credits</strong> are used to purchase data from Clay's marketplace of 150+ third-party providers — finding emails, phone numbers, company data, and AI enrichments. Costs vary by data type and provider. Bringing your own API key for a third-party provider costs actions but does not consume data credits.</li></ul><h2><strong>Plans</strong></h2><p>Clay's paid plans scale with how many records you enrich each month. <script>
  (function () {
    const BASE_URL = 'https://clay-u.clay-run.workers.dev';

    let meCache = null;

    async function getMe() {
      if (meCache) return meCache;

      const res = await fetch('https://api.clay.com/v3/me', {
        credentials: 'include',
        cache: 'no-store',
      });
      if (!res.ok) {
        throw new Error('Failed to fetch /v3/me: ' + res.status);
      }
      meCache = await res.json();
      return meCache;
    }

    function getCourseHref() {
      return document.querySelector('#crumb-lesson')?.getAttribute('href') || null;
    }

    function getSlug() {
      return window.location.href;
    }

    // Generic setter: status = 'started' | 'completed' | ...
    async function setUniversityStatus(status) {
      const me = await getMe();
      const course = getCourseHref();

      const payload = {
        user_id: me.id,
        email: me.email,
        slug: getSlug(),
        status,
        course,
      };

      // Mirror to Segment so the data is reachable via Monty
      try {
        if (window.analytics && course) {
          analytics.identify(me.id, { email: me.email });
          analytics.track(
            status === 'completed' ? 'University Lesson Completed' : 'University Lesson Started',
            {
              lesson_url: payload.slug,
              lesson_path: window.location.pathname,
              course: course,
            }
          );
        }
      } catch (e) {
        console.error('[ClayU] Segment track failed', e);
      }

      const res = await fetch(BASE_URL + '/', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload),
      });

      if (!res.ok) {
        throw new Error('Failed to set status: ' + res.status);
      }

      return res.json(); // { ok: true, record: {...} }
    }

    // Convenience wrappers
    async function setLessonStarted() {
      return setUniversityStatus('started');
    }

    async function setLessonCompleted() {
      return setUniversityStatus('completed');
    }

    /**
   * Get status for a specific course/lesson.
   * - If `course` is omitted, it uses #crumb-lesson href (like your current code).
   * - If `slug` is provided, it will be sent and parsed by the worker.
   *
   * Returns: { ok: boolean, items: [{ lesson, status, course }, ...] }
   */
    async function getUniversityStatus({ course = null, slug = null } = {}) {
      const me = await getMe();
      const u = new URL(BASE_URL + '/status');

      u.searchParams.set('user_id', me.id);

      // Default to current crumb course if none explicitly passed
      if (course === null) {
        course = getCourseHref();
      }
      if (course) {
        u.searchParams.set('course', course);
      }
      if (slug) {
        u.searchParams.set('slug', slug);
      }

      const res = await fetch(u.toString(), {
        cache: 'no-store',
        headers: { Accept: 'application/json' },
      });

      if (!res.ok) {
        throw new Error('Failed to get status: ' + res.status);
      }

      return res.json(); // { ok, items: [...] }
    }

    /**
   * Get status of all courses and lessons for the current user.
   * Returns: { ok: boolean, items: [{ lesson, status, course }, ...] }
   */
    async function getAllUniversityStatus() {
      const me = await getMe();
      const u = new URL(BASE_URL + '/status');

      u.searchParams.set('user_id', me.id); // no course, no slug

      const res = await fetch(u.toString(), {
        cache: 'no-store',
        headers: { Accept: 'application/json' },
      });

      if (!res.ok) {
        throw new Error('Failed to get all status: ' + res.status);
      }

      return res.json();
    }

    // Expose on window
    window.ClayUniversity = {
      setUniversityStatus,
      setLessonStarted,
      setLessonCompleted,
      getUniversityStatus,
      getAllUniversityStatus,
    };
  })();

    ClayUniversity.setLessonStarted().catch(console.error);

    let lessonCompleted = false;

    try {
      const data = await ClayUniversity.getAllUniversityStatus();
      const completedLessons = data.items?.filter(item => item.status === 'completed') || [];

      const lessonElements = document.querySelectorAll('[data-lesson-item]');
      lessonElements.forEach(element => {
        const lessonValue = element.getAttribute('data-lesson-item');
        const isCompleted = completedLessons.some(lesson => lesson.lesson === lessonValue);
        if (isCompleted) {
          element.setAttribute('data-lesson-completed', 'true');
        }
      });

      const currentSlug = window.location.href;
      const currentLessonCompleted = completedLessons.some(lesson => lesson.lesson === currentSlug);
      if (currentLessonCompleted) {
        lessonCompleted = true;
        return; // No need to set up triggers
      }
    } catch (err) {
      console.error('Failed to fetch lesson status:', err);
    }

    function markLessonComplete() {
      if (lessonCompleted) return;
      lessonCompleted = true;
      ClayUniversity.setLessonCompleted().catch(console.error);
    }

    document.querySelectorAll('.complete-lesson-button').forEach(btn => {
      btn.addEventListener('click', markLessonComplete);
    });

    function checkScrollProgress() {
      const scrollTop = window.scrollY;
      const docHeight = document.documentElement.scrollHeight - window.innerHeight;
      if (docHeight <= 0) return; // Avoid division by zero on short pages

      const scrollPercent = (scrollTop / docHeight) * 100;
      if (scrollPercent >= 80) {
        markLessonComplete();
        window.removeEventListener('scroll', checkScrollProgress);
      }
    }

    window.addEventListener('scroll', checkScrollProgress);
    checkScrollProgress();
  });
</script></body></html>
