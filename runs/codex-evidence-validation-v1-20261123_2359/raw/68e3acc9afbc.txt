  (function () {
    var AUTH_EVENTS = {
      success: "clay:auth:success",
      failure: "clay:auth:failure",
    };
    var AUTH_STATE_KEY = "__clayAuthState";

    var revealPage = function () {
      const wrapper = document.querySelector(".page-wrapper");
      if (wrapper) wrapper.style.visibility = "visible";
    };

    var setDisplay = function (selector, value) {
      document.querySelectorAll(selector).forEach(function (el) {
        el.style.display = value;
      });
    };

    var updateState = function (state) {
      window[AUTH_STATE_KEY] = state;
    };

    var dispatchAuthEvent = function (type, detail) {
      try {
        document.dispatchEvent(new CustomEvent(type, { detail: detail }));
      } catch (err) {
        console.error("[Clay] Failed to dispatch auth event:", err);
      }
    };

    var showLoggedIn = function () {
      setDisplay('[logged-in="true"]', "flex");
      setDisplay('[logged-in="false"]', "none");
    };

    var showLoggedOut = function () {
      setDisplay('[logged-in="true"]', "none");
      setDisplay('[logged-in="false"]', "flex");
    };

    document.addEventListener("DOMContentLoaded", async function () {
      updateState({ status: "loading" });
      showLoggedOut(); // default until we know

      try {
        var response = await fetch("https://api.clay.com/v3/me", {
          credentials: "include",
        });

        if (!response.ok) throw new Error("Not logged in");

        var user = await response.json();
        if (!user?.id) throw new Error("Missing user data");

        showLoggedIn();
        updateState({ status: "authenticated", user: user });
        dispatchAuthEvent(AUTH_EVENTS.success, { user: user });

      } catch (err) {
        showLoggedOut();
        updateState({ status: "unauthenticated", error: err });
        dispatchAuthEvent(AUTH_EVENTS.failure, { error: err });

      } finally {
        revealPage();
      }
    });
  })();

<script>
  //---- Modal JS ----
  // Get all dialog elements
  const dialogs = document.querySelectorAll("dialog");
  // Get the corresponding show buttons for each dialog
  const showButtons = document.querySelectorAll("dialog + button");
  // Get the corresponding close buttons for each dialog
  const closeButtons = document.querySelectorAll("dialog button");

  // Loop through each dialog and add event listeners
  dialogs.forEach((dialog, index) => {
    // "Show the dialog" button opens the dialog modally
    showButtons[index].addEventListener("click", () => {
      dialog.showModal();
    });

    // "Close" button closes the dialog
    closeButtons[index].addEventListener("click", () => {
      dialog.close();
    });

    // Close the dialog when clicking outside of it
    dialog.addEventListener("click", (e) => {
      const dialogDimensions = dialog.getBoundingClientRect();
      if (
        e.clientX < dialogDimensions.left ||
        e.clientX > dialogDimensions.right ||
        e.clientY < dialogDimensions.top ||
        e.clientY > dialogDimensions.bottom
      ) {
        dialog.close();
      }
    });
  });

  //---- Update footer year ----
  document.addEventListener('DOMContentLoaded', function() {
    const elements = document.querySelectorAll('.year');
    elements.forEach(element => {
      element.textContent = new Date().getFullYear();
    });
  });
</script>

<script>
  (function () {
    const BASE_URL = 'https://clay-u.clay-run.workers.dev';

    let meCache = null;

    async function getMe() {
      if (meCache) return meCache;

      const res = await fetch('https://api.clay.com/v3/me', {
        credentials: 'include',
        cache: 'no-store',
      });
      if (!res.ok) {
        throw new Error('Failed to fetch /v3/me: ' + res.status);
      }
      meCache = await res.json();
      return meCache;
    }

    function getCourseHref() {
      return document.querySelector('#crumb-lesson')?.getAttribute('href') || null;
    }

    function getSlug() {
      return window.location.href;
    }

    // Generic setter: status = 'started' | 'completed' | ...
    async function setUniversityStatus(status) {
      const me = await getMe();
      const course = getCourseHref();

      const payload = {
        user_id: me.id,
        email: me.email,
        slug: getSlug(),
        status,
        course,
      };

      // Mirror to Segment so the data is reachable via Monty
      try {
        if (window.analytics && course) {
          analytics.identify(me.id, { email: me.email });
          analytics.track(
            status === 'completed' ? 'University Lesson Completed' : 'University Lesson Started',
            {
              lesson_url: payload.slug,
              lesson_path: window.location.pathname,
              course: course,
            }
          );
        }
      } catch (e) {
        console.error('[ClayU] Segment track failed', e);
      }

      const res = await fetch(BASE_URL + '/', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload),
      });

      if (!res.ok) {
        throw new Error('Failed to set status: ' + res.status);
      }

      return res.json(); // { ok: true, record: {...} }
    }

    // Convenience wrappers
    async function setLessonStarted() {
      return setUniversityStatus('started');
    }

    async function setLessonCompleted() {
      return setUniversityStatus('completed');
    }

    /**
   * Get status for a specific course/lesson.
   * - If `course` is omitted, it uses #crumb-lesson href (like your current code).
   * - If `slug` is provided, it will be sent and parsed by the worker.
   *
   * Returns: { ok: boolean, items: [{ lesson, status, course }, ...] }
   */
    async function getUniversityStatus({ course = null, slug = null } = {}) {
      const me = await getMe();
      const u = new URL(BASE_URL + '/status');

      u.searchParams.set('user_id', me.id);

      // Default to current crumb course if none explicitly passed
      if (course === null) {
        course = getCourseHref();
      }
      if (course) {
        u.searchParams.set('course', course);
      }
      if (slug) {
        u.searchParams.set('slug', slug);
      }

      const res = await fetch(u.toString(), {
        cache: 'no-store',
        headers: { Accept: 'application/json' },
      });

      if (!res.ok) {
        throw new Error('Failed to get status: ' + res.status);
      }

      return res.json(); // { ok, items: [...] }
    }

    /**
   * Get status of all courses and lessons for the current user.
   * Returns: { ok: boolean, items: [{ lesson, status, course }, ...] }
   */
    async function getAllUniversityStatus() {
      const me = await getMe();
      const u = new URL(BASE_URL + '/status');

      u.searchParams.set('user_id', me.id); // no course, no slug

      const res = await fetch(u.toString(), {
        cache: 'no-store',
        headers: { Accept: 'application/json' },
      });

      if (!res.ok) {
        throw new Error('Failed to get all status: ' + res.status);
      }

      return res.json();
    }

    // Expose on window
    window.ClayUniversity = {
      setUniversityStatus,
      setLessonStarted,
      setLessonCompleted,
      getUniversityStatus,
      getAllUniversityStatus,
    };
  })();

  document.addEventListener('DOMContentLoaded', async () => {
    ClayUniversity.setLessonStarted().catch(console.error);

    let lessonCompleted = false;

    try {
      const data = await ClayUniversity.getAllUniversityStatus();
      const completedLessons = data.items?.filter(item => item.status === 'completed') || [];

      const lessonElements = document.querySelectorAll('[data-lesson-item]');
      lessonElements.forEach(element => {
        const lessonValue = element.getAttribute('data-lesson-item');
        const isCompleted = completedLessons.some(lesson => lesson.lesson === lessonValue);
        if (isCompleted) {
          element.setAttribute('data-lesson-completed', 'true');
        }
      });

      const currentSlug = window.location.href;
      const currentLessonCompleted = completedLessons.some(lesson => lesson.lesson === currentSlug);
      if (currentLessonCompleted) {
        lessonCompleted = true;
        return; // No need to set up triggers
      }
    } catch (err) {
      console.error('Failed to fetch lesson status:', err);
    }

    function markLessonComplete() {
      if (lessonCompleted) return;
      lessonCompleted = true;
      ClayUniversity.setLessonCompleted().catch(console.error);
    }

    document.querySelectorAll('.complete-lesson-button').forEach(btn => {
      btn.addEventListener('click', markLessonComplete);
    });

    function checkScrollProgress() {
      const scrollTop = window.scrollY;
      const docHeight = document.documentElement.scrollHeight - window.innerHeight;
      if (docHeight <= 0) return; // Avoid division by zero on short pages

      const scrollPercent = (scrollTop / docHeight) * 100;
      if (scrollPercent >= 80) {
        markLessonComplete();
        window.removeEventListener('scroll', checkScrollProgress);
      }
    }

    window.addEventListener('scroll', checkScrollProgress);
    checkScrollProgress();
  });
</script></body></html>
