    return 'desktop';
  };

  const getOrderAttribute = (el, deviceType) => {
    if (deviceType === 'mobile') {
      return el.closest('[ul-scroll-order="mobile-first"]') ? 'first' :
             el.closest('[ul-scroll-order="mobile-last"]') ? 'last' :
             el.closest('[ul-scroll-order="tablet-first"]') ? 'first' :

    }

             el.closest('[ul-scroll-order="tablet-last"]') ? 'last' :
             'middle';
    }
    return el.closest('[ul-scroll-order="desktop-first"]') ? 'first' :
           el.closest('[ul-scroll-order="desktop-last"]') ? 'last' :

  const animateElements = (container) => {

    const elements = [
      ...container.querySelectorAll('[ul-scroll-el]:not([ul-scroll-el="none"])'),

      middle: [],
      last: []
    };

    elements.forEach(el => {
      const order = getOrderAttribute(el, deviceType);
      orderMap[order].push(el);
    });

    let index = 0;
    ['first', 'middle', 'last'].forEach(group => {
      orderMap[group].forEach(el => {
